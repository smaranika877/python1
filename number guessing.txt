import random

def number_guessing_game():
    print("Hey buddy, welcome to the number guessing game!")
    print("Start")
    print("I'm thinking of a number between 1 to 100.")

    # Generate a random number and initialize attempts
    target_number = random.randint(1, 100)
    attempts = 0

    while True:
        try:
            # Get user input and increment attempts
            guess = int(input("Make a guess: "))
            attempts += 1

            if guess < target_number:
                print("Too low")
            elif guess > target_number:
                print("Too high")
            else:
                print(f"Congratulations! You've guessed the number {target_number} in {attempts} attempts!")
                break
        except ValueError:
            print("Please enter a valid number.")

# Run the game
number_guessing_game()
